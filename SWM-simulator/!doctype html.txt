    <script>
      let chart;
      lucide.createIcons();

      function updateStats(bins) {
        const avgFill =
          bins.reduce((acc, b) => acc + b.fill_level, 0) / bins.length;
        document.getElementById("avgFillLevel").innerText =
          `${avgFill.toFixed(1)}%`;

        const alertEl = document.getElementById("alertSummary");
        const critical = bins.filter((b) => b.fill_level > 80);
        if (critical.length > 0) {
          alertEl.innerText = `${critical.length} Critical`;
        } else {
          alertEl.innerText = "All Clear";
        }
      }

      function renderBins(nodes) {
        const container = document.getElementById("binsContainer");
        container.innerHTML = "";

        nodes.forEach((node) => {
          const isActive = node.type === "active";
          const value = isActive ? node.fill_level : 0;

          let colors = {
            start: "#d1d5db",
            end: "#9ca3af",
            glow: "rgba(156,163,175,0.25)",
          }; // default gray for empty/offline

          if (node.bin_id === "Bin_1" || node.bin_id === "Organic") {
            colors = {
              start: "#FFFFFF",
              end: "#059669",
              glow: "rgba(16,185,129,0.25)",
            };
          } else if (node.bin_id === "Bin_2" || node.bin_id === "Plastic") {
            colors = {
              start: "#FFFFFF",
              end: "#f59e0b",
              glow: "rgba(245,158,11,0.25)",
            };
          } else if (node.bin_id === "Bin_3" || node.bin_id === "Metal") {
            colors = {
              start: "#FFFFFF",
              end: "#e11d48",
              glow: "rgba(244,63,94,0.25)",
            };
          }
          const card = `
                <div class="flex flex-col items-center group transition-all duration-700 ${!isActive ? "opacity-30" : ""}">
                  <div class="relative w-28 h-56 bg-white border-[4px] border-white rounded-[1.5rem] overflow-hidden shadow-2xl ring-1 ring-slate-100 bin-glow transition-all duration-500" style="--glow-color: ${isActive ? colors.glow : "transparent"}">
                    ${isActive ? `<div class="wave" style="height:${value}%; --wave-color-start:${colors.start}; --wave-color-end:${colors.end}"></div>` : ""}
                    <div class="absolute inset-0 flex flex-col items-center justify-center pointer-events-none">
                      ${
                        isActive
                          ? `<span class="text-4xl font-display font-bold text-slate-900 drop-shadow-sm">${value.toFixed(0)}</span><span class="text-[8px] font-bold uppercase tracking-[0.2em] text-slate-500/80">Percent</span>`
                          : `<i data-lucide="shield-off" class="w-8 h-8 text-slate-200 mb-2"></i><span class="text-[8px] font-bold uppercase tracking-widest text-slate-300">Offline</span>`
                      }
                    </div>
                  </div>
                  <div class="mt-6 text-center">
                    <p class="text-lg font-display font-bold text-slate-900 tracking-tight">${node.bin_id}</p>
                  </div>
                </div>`;
          container.innerHTML += card;
        });
        lucide.createIcons();
      }

      function updatePriority(data) {
        const container = document.getElementById("priorityList");
        container.innerHTML = "";
        data.forEach((bin) => {
          const isCritical = bin.fill_level > 80;
          container.innerHTML += `<div class="flex items-center justify-between"><span class="text-xs font-bold text-slate-700">${bin.bin_id}</span><span class="text-xs font-bold ${isCritical ? "text-rose-500" : "text-indigo-600"}">${bin.fill_level.toFixed(0)}%</span></div>`;
        });
      }

      function createChart(labels, values) {
        const ctx = document.getElementById("fillChart").getContext("2d");
        const gradient = ctx.createLinearGradient(0, 0, 0, 400);
        gradient.addColorStop(0, "rgba(99, 102, 241, 0.15)");
        gradient.addColorStop(1, "rgba(99, 102, 241, 0)");
        chart = new Chart(ctx, {
          type: "line",
          data: {
            labels,
            datasets: [
              {
                data: values,
                borderColor: "#6366f1",
                borderWidth: 6,
                pointBackgroundColor: "#ffffff",
                pointBorderColor: "#6366f1",
                pointBorderWidth: 3,
                pointRadius: 0,
                pointHoverRadius: 6,
                fill: true,
                backgroundColor: gradient,
              },
            ],
          },
          options: {
            responsive: true,
            plugins: { legend: { display: false } },
            interaction: { intersect: false, mode: "index" },
            scales: {
              x: { display: false },
              y: { display: true, max: 100, min: 0 },
            },
          },
        });
      }
      // ===== Weekly Wastage Charts =====
      function createWeeklyChart(ctx, color) {
        return new Chart(ctx, {
          type: "bar",
          data: {
            labels: ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"],
            datasets: [
              {
                label: "Waste (kg)",
                data: [0, 0, 0, 0, 0, 0, 0],
                backgroundColor: color,
                borderRadius: 6,
              },
            ],
          },
          options: {
            responsive: true,
            plugins: { legend: { display: false } },
            scales: {
              y: { beginAtZero: true, max: 20 },
              x: { grid: { display: false } },
            },
          },
        });
      }

      // Get canvas elements
      const organicCtx = document
        .getElementById("weeklyOrganicChart")
        .getContext("2d");
      const plasticCtx = document
        .getElementById("weeklyPlasticChart")
        .getContext("2d");
      const metalCtx = document
        .getElementById("weeklyMetalChart")
        .getContext("2d");

      // Initialize charts
      const organicChart = createWeeklyChart(organicCtx, "rgba(34,197,94,0.7)"); // green
      const plasticChart = createWeeklyChart(

      // Firebase Listener
      const ref = db.ref("dustbin"); // your root path
const chartLabels = [];
const chartValues = [];

ref.on("value", (snapshot) => {
  const data = snapshot.val();
  console.log("Firebase snapshot:", data); // DEBUG

  if (!data) return;

  // ─── Only Organic bin has real-time data ───
  const organicFill = parseFloat(data.filled_percentage || 0);

  const bins = [
    { bin_id: "Organic", fill_level: organicFill, type: "active", timestamp: Date.now() },
    { bin_id: "Plastic", fill_level: 0, type: "empty", timestamp: Date.now() },
    { bin_id: "Metal", fill_level: 0, type: "empty", timestamp: Date.now() },
  ];

  // ─── Update alerts for Organic only ───
  const alertEl = document.getElementById("alertSummary");
  if (bins[0].fill_level > 90) {
    alertEl.innerText = "🚨 Bin Critically Full!";
    if (alarmEnabled) alarm.play();
  } else if (bins[0].fill_level > 80) {
    alertEl.innerText = "⚠️ Bin Full!";
    if (alarmEnabled) alarm.play();
  } else {
    alertEl.innerText = "All Clear";
    alarm.pause();
    alarm.currentTime = 0;
  }

  // ─── Always render all bins ───
  renderBins(bins);
  updatePriority(bins);
  updateStats(bins);

  // ─── Update chart only for Organic bin ───
  chartLabels.push(new Date().toLocaleTimeString());
  chartValues.push(bins[0].fill_level);
  if (chartLabels.length > 20) {
    chartLabels.shift();
    chartValues.shift();
  }
  if (!chart) createChart(chartLabels, chartValues);
  else {
    chart.data.labels = chartLabels;
    chart.data.datasets[0].data = chartValues;
    chart.update();
  }

  // ─── Update last timestamp ───
  document.getElementById("lastUpdate").innerText = new Date().toLocaleTimeString();
});
